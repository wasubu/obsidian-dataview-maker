3. JavaScript Logic

// --- JS ---
container.innerHTML = htmlContent;

let books = [];
try {
  if (typeof db !== 'undefined' && db.query) {
    books = db.query('"Books"').map(b => ({
      title: b.title || b.file?.name || "Untitled",
      author: b.author || "Unknown Author",
      cover: b.cover || "https://via.placeholder.com/150x225?text=No+Cover",
      status: b.status || "Plan to Read",
      path: b.file?.path || '#'
    }));
  }
} catch (err) {}

if (books.length === 0) {
  books = [
    { title: "Atomic Habits", author: "James Clear", cover: "https://via.placeholder.com/150x225?text=Atomic+Habits", status: "Reading", path: "#" }
  ];
}

const filterBar = container.querySelector('#filter-bar-container');
const addBtn = container.querySelector('#add-book-btn');
const gridContainer = container.querySelector('#book-grid-container');

if (addBtn) {
  addBtn.onclick = async () => {
    await db.createNote({
      folder: "Books",
      title: "New Book",
      frontmatter: { title: "New Book", author: "Unknown Author", status: "Plan to Read", cover: "" },
      content: "# New Book\n\n",
      open: true
    });
  };
}

const categories = ["All", "Reading", "Completed", "Plan to Read"];
let activeCategory = "All";

function renderGrid(category) {
  if (!gridContainer) return;
  gridContainer.innerHTML = "";

  const filtered = books.filter(b => category === "All" || (b.status || "").toLowerCase().includes(category.toLowerCase()));

  if (filtered.length === 0) {
    gridContainer.innerHTML = '<div style="grid-column:1/-1;text-align:center;padding:30px;color:var(--text-muted);">No books found</div>';
    return;
  }

  for (let book of filtered) {
    const card = document.createElement('div');
    card.className = "book-card";
    card.onclick = () => {
      if (book.path !== "#" && db.openNote) db.openNote(book.path);
      else db.notify(`Clicked: ${book.title}`);
    };
    card.innerHTML = `
      <img class="book-cover" src="${book.cover}" alt="${book.title}" loading="lazy">
      <div class="book-info">
        <div class="book-title">${book.title}</div>
        <div class="book-author">${book.author}</div>
        <div class="book-status">${book.status}</div>
      </div>
    `;
    gridContainer.appendChild(card);
  }
}

if (filterBar) {
  db.bindFilter('#filter-bar-container', categories, (selected) => {
    activeCategory = selected;
    renderGrid(selected);
  });
}

renderGrid("All");
