Tutorial: Habit Tracker & State Management

Overview

This tutorial demonstrates how to build an interactive Habit Tracker & Pomodoro Timer dashboard using advanced runtime helpers like $ and $$ for DOM selection, db.state() for reactive UI updates, db.storage() for persistent local storage, db.timer() for countdown intervals, and db.modal() for popup forms.

Core Concept: $ and $$ are automatically scoped to your canvas container, meaning they will never interfere with other elements on your Obsidian page.

3. JavaScript Logic

// --- JS ---
container.innerHTML = htmlContent;

// 1. Persistent Storage & Reactive State Store
const habitStore = db.storage('habit_tracker');
const habitsState = db.state(habitStore.get('habits', [
  { id: 1, title: "Drink Water", streak: 5, done: false },
  { id: 2, title: "Workout 30m", streak: 3, done: true },
  { id: 3, title: "Read 10 Pages", streak: 12, done: false }
]), (updatedHabits) => {
  habitStore.set('habits', updatedHabits);
  renderHabits();
});

// 2. DOM Selection using $ helper
const habitListEl = $('#habit-list');
const addHabitBtn = $('#add-habit-btn');
const timerDisplay = $('#timer-display');
const startTimerBtn = $('#start-timer-btn');

// 3. Managed Timer & Audio Synthesis
let timeLeft = 25 * 60;
const pomodoroTimer = db.timer(1000, () => {
  if (timeLeft > 0) {
    timeLeft--;
    const mins = Math.floor(timeLeft / 60).toString().padStart(2, '0');
    const secs = (timeLeft % 60).toString().padStart(2, '0');
    timerDisplay.innerText = `${mins}:${secs}`;
  } else {
    pomodoroTimer.stop();
    db.beep(880, 0.5, 'sine', 0.2);
    db.notify("Pomodoro session complete! Take a break.");
  }
});

startTimerBtn.onclick = () => {
  if (pomodoroTimer.isRunning) {
    pomodoroTimer.stop();
    startTimerBtn.innerText = "Start Pomodoro";
  } else {
    pomodoroTimer.start();
    startTimerBtn.innerText = "Pause Pomodoro";
    db.notify("Pomodoro started (25:00)");
  }
};

// 4. Native DOM Modal Popup
addHabitBtn.onclick = () => {
  db.modal("Add New Habit", (bodyEl, close) => {
    bodyEl.innerHTML = `
      <div style="display:flex;flex-direction:column;gap:12px;">
        <label style="font-size:0.85rem;font-weight:600;">Habit Name:</label>
        <input type="text" id="new-habit-input" placeholder="e.g. Meditate" style="padding:8px;border-radius:6px;border:1px solid var(--background-modifier-border);background:var(--background-secondary);color:var(--text-normal);outline:none;">
        <button id="save-habit-btn" style="background:var(--interactive-accent);color:var(--text-on-accent);border:none;padding:8px;border-radius:6px;cursor:pointer;font-weight:600;">Save Habit</button>
      </div>
    `;
    bodyEl.querySelector('#save-habit-btn').onclick = () => {
      const val = bodyEl.querySelector('#new-habit-input').value.trim();
      if (val) {
        habitsState.update(prev => [
          ...prev,
          { id: Date.now(), title: val, streak: 0, done: false }
        ]);
        db.notify(`Created habit: ${val}`);
        close();
      }
    };
  });
};

// 5. Render List using db.renderList helper
function renderHabits() {
  db.renderList('#habit-list', habitsState.val, (habit) => `
    <div class="habit-card ${habit.done ? 'completed' : ''}" data-id="${habit.id}">
      <div class="habit-info">
        <div class="habit-title">${habit.title}</div>
        <div class="habit-streak">🔥 ${habit.streak} day streak</div>
      </div>
      <button class="check-btn" onclick="toggleHabit(${habit.id})">${habit.done ? '✅ Done' : '⭕ Mark Done'}</button>
    </div>
  `, '<div class="no-habits">No habits yet. Click "+ Add Habit" to start!</div>');
}

// Global toggle function for inline click handling
window.toggleHabit = (id) => {
  habitsState.update(prev => prev.map(h => {
    if (h.id === id) {
      const nextDone = !h.done;
      if (nextDone) {
        db.beep(523, 0.1, 'sine', 0.1);
        db.notify(`Completed "${h.title}"! 🎉`);
        return { ...h, done: nextDone, streak: h.streak + 1 };
      }
      return { ...h, done: nextDone };
    }
    return h;
  }));
};

renderHabits();
