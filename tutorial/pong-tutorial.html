Tutorial: Building a Retro Pong Game

Overview

Custom Database Studio isn't just for static databases—because it supports full HTML5 Canvas and 60FPS requestAnimationFrame loops, you can build fully interactive games directly inside your Obsidian notes!

Key APIs Used: db.canvas() for high-DPI scaling, db.loop() for game physics, and db.beep() for procedural retro sound effects.

3. JavaScript Game Logic

// --- JS ---
container.innerHTML = htmlContent;

const setup = db.canvas('#pongCanvas');
if (!setup) return;
const { canvas, ctx, width, height, clear } = setup;

let playerY = height / 2 - 40;
let aiY = height / 2 - 40;
const paddleWidth = 12;
const paddleHeight = 80;

let ballX = width / 2;
let ballY = height / 2;
let ballSpeedX = 4;
let ballSpeedY = 3;
const ballSize = 10;

let playerScore = 0;
let aiScore = 0;

let upPressed = false;
let downPressed = false;

window.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') upPressed = true;
  if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') downPressed = true;
});

window.addEventListener('keyup', (e) => {
  if (e.key === 'ArrowUp' || e.key === 'w' || e.key === 'W') upPressed = false;
  if (e.key === 'ArrowDown' || e.key === 's' || e.key === 'S') downPressed = false;
});

const gameLoop = db.loop((delta, time) => {
  if (upPressed) playerY -= 6;
  if (downPressed) playerY += 6;
  playerY = Math.max(0, Math.min(height - paddleHeight, playerY));

  const aiCenter = aiY + paddleHeight / 2;
  if (aiCenter < ballY - 15) aiY += 4;
  else if (aiCenter > ballY + 15) aiY -= 4;
  aiY = Math.max(0, Math.min(height - paddleHeight, aiY));

  ballX += ballSpeedX;
  ballY += ballSpeedY;

  if (ballY <= 0 || ballY >= height - ballSize) {
    ballSpeedY *= -1;
    db.beep(300, 0.05, 'square', 0.05);
  }

  if (ballX <= 30 + paddleWidth && ballY >= playerY && ballY <= playerY + paddleHeight) {
    ballSpeedX *= -1;
    db.beep(600, 0.08, 'sine', 0.1);
  }

  if (ballX >= width - 30 - paddleWidth && ballY >= aiY && ballY <= aiY + paddleHeight) {
    ballSpeedX *= -1;
    db.beep(600, 0.08, 'sine', 0.1);
  }

  if (ballX < 0) {
    aiScore++;
    db.beep(150, 0.2, 'sawtooth', 0.1);
    ballX = width / 2;
    ballY = height / 2;
    ballSpeedX = 4;
  } else if (ballX > width) {
    playerScore++;
    db.beep(800, 0.2, 'sine', 0.1);
    ballX = width / 2;
    ballY = height / 2;
    ballSpeedX = -4;
  }

  clear('#11111b');

  ctx.fillStyle = '#cba6f7';
  for (let i = 0; i < height; i += 20) {
    ctx.fillRect(width / 2 - 1, i, 2, 10);
  }

  ctx.fillRect(30, playerY, paddleWidth, paddleHeight);
  ctx.fillRect(width - 30 - paddleWidth, aiY, paddleWidth, paddleHeight);
  ctx.fillRect(ballX, ballY, ballSize, ballSize);

  ctx.font = '32px monospace';
  ctx.fillStyle = '#cdd6f4';
  ctx.fillText(playerScore, width / 4, 50);
  ctx.fillText(aiScore, (width / 4) * 3, 50);
});

gameLoop.start();
